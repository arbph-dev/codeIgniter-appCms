```
// ListPanelBase extends PanelBase

// --- callbacks (Workbench → enregistre) ---
onSearch(fn)  // fn(q: string)
onSelect(fn)  // fn(item)
onNew(fn)     // fn()
onPage(fn)    // fn(page: number)

// --- Workbench → Panel ---
show(items, pager = null)   // sous-classe : table ; base : pager via _renderPager
showLoading()
showError(msg)
clear()
destroy()                   // nullifie callbacks + super

// --- hooks sous-classe ---
_renderToolbar()            // optionnel
_renderRows(items)          // obligatoire — retourne ou remplit tableEl
_getPaginationOptions()     // { style, maxVisible } si besoin

//

_renderPager(pager) {
    clear(this.pagerEl)
    if (!pager) return
    this.pagerEl.appendChild(
        pagination({
            pager,
            onClick    : (page) => this._onPageFn?.(page),
            style      : this._pagerStyle ?? 'compact',
            maxVisible : this._pagerMaxVisible ?? 5,
        })
    )
}

```
### OrgListPanel V2

```
// assets/js/ui/workbench/organisation/OrgListPanel2.js

import ListPanelBase from '/assets/js/ui/workbench/core/ListPanelBase.js'
import { table } from '/assets/js/core/domhelper.js'

export class OrgListPanel extends ListPanelBase
{
    constructor()
    {
        super({
            title             : 'Organisations',
            newLabel          : '+ Nouveau',
            panelClass        : 'wb_mot_list_panel', // ou wb_org_list_panel
            searchPlaceholder : 'Nom, SIREN…',
            pagerStyle        : 'compact',
            pagerMaxVisible   : 5,
        })
    }

    _renderRows(items)
    {
        this.tableEl.appendChild(
            table({
                data    : items,
                columns : [
                    { key: 'nom',        label: 'Nom' },
                    { key: 'type_label', label: 'Type' },
                    { key: 'siren',      label: 'SIREN' },
                ],
                attrs      : { class: 'cp_table' },
                onRowClick : (row) => this._onSelectFn?.(row),
            })
        )
    }
}

export default OrgListPanel
```
