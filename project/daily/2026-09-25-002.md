### article

Je modifierais donc la migration ainsi :

```sql
CREATE TABLE IF NOT EXISTS article
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    lbl VARCHAR(255) NOT NULL,
    `desc` TEXT NULL,

    -- Organisation fabricant
    manid BIGINT UNSIGNED NULL,

    mdl VARCHAR(255) NULL,
    ean VARCHAR(32) NULL,

    -- Type fonctionnel de l'article
    typid BIGINT UNSIGNED NULL,

    ref VARCHAR(255) NULL,

    stock DECIMAL(15,3) NOT NULL DEFAULT 0,
    stockmini DECIMAL(15,3) NOT NULL DEFAULT 0,
    stockmaxi DECIMAL(15,3) NULL,

    unite VARCHAR(20) NULL,
    emplacement VARCHAR(255) NULL,
    actif TINYINT(1) NOT NULL DEFAULT 1,

    -- Zone d'extension pour les données encore non normalisées
    data JSON NULL,

    created_at DATETIME NULL,
    updated_at DATETIME NULL,

    PRIMARY KEY (id),

    KEY idx_article_lbl (lbl),
    KEY idx_article_ref (ref),
    KEY idx_article_ean (ean),
    KEY idx_article_manid (manid),
    KEY idx_article_typid (typid),
    KEY idx_article_actif (actif),
    KEY idx_article_stock (stock),

    CONSTRAINT fk_article_manid
        FOREIGN KEY (manid)
        REFERENCES organisations(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_article_typid
        FOREIGN KEY (typid)
        REFERENCES articletype(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;
```


### articlefournisseur
Le fabricant ou fournisseur est une **organisation**, l'établissement représente une implantation précise.
> « Le relais est acheté chez Rexel Rennes et non simplement chez Rexel »
> a prévoir etbid  → etablissements.id

```sql
CREATE TABLE IF NOT EXISTS articlefournisseur
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    artid BIGINT UNSIGNED NOT NULL,

    -- Organisation fournisseur
    fourid BIGINT UNSIGNED NOT NULL,

    ref_four VARCHAR(255) NULL,
    prix DECIMAL(15,2) NULL,
    devise VARCHAR(3) NOT NULL DEFAULT 'EUR',
    delai INT NULL,
    actif TINYINT(1) NOT NULL DEFAULT 1,

    PRIMARY KEY (id),

    KEY idx_articlefour_artid (artid),
    KEY idx_articlefour_fourid (fourid),
    KEY idx_articlefour_actif (actif),

    CONSTRAINT fk_articlefour_article
        FOREIGN KEY (artid)
        REFERENCES article(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_articlefour_organisation
        FOREIGN KEY (fourid)
        REFERENCES organisations(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;
```

Et à côté :

```
organisations
      │
      ├── entreprises
      │

```

Cette séparation est beaucoup plus propre que notre premier script et surtout elle évite de créer une deuxième représentation des entreprises dans la GMAO.

### Un autre avantage important

Un même fournisseur peut maintenant être utilisé ailleurs dans l'application :

```
organisations
    └── Rexel
         ├── organisation
         ├── entreprise
         ├── établissements
         ├── articles fournisseur
         ├── interventions prestataire
         └── autres relations métier
```

On obtient donc réellement un **référentiel transversal**, ce qui est exactement le rôle d'`organisations`.

```sql
-- =====================================================================
-- GMAO - MODULE ARTICLES
-- Migration initiale MySQL
-- =====================================================================

SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;


-- =====================================================================
-- 1. FABRICANTS **** OBSOLETE VOIR PLUS HAUT
-- =====================================================================

CREATE TABLE IF NOT EXISTS manufact
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    lbl VARCHAR(255) NOT NULL,
    web VARCHAR(255) NULL,
    pays VARCHAR(100) NULL,
    contact VARCHAR(255) NULL,
    tel VARCHAR(50) NULL,
    email VARCHAR(255) NULL,
    notes TEXT NULL,

    PRIMARY KEY (id),
    KEY idx_manufact_lbl (lbl)
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 2. TYPES D'ARTICLES
-- =====================================================================

CREATE TABLE IF NOT EXISTS articletype
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    lbl VARCHAR(255) NOT NULL,
    `desc` TEXT NULL,
    atpid BIGINT UNSIGNED NULL,

    PRIMARY KEY (id),
    KEY idx_articletype_lbl (lbl),
    KEY idx_articletype_parent (atpid),

    CONSTRAINT fk_articletype_parent
        FOREIGN KEY (atpid)
        REFERENCES articletype(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 3. CARACTÉRISTIQUES
-- =====================================================================

CREATE TABLE IF NOT EXISTS caracteristique
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    lbl VARCHAR(255) NOT NULL,
    unite VARCHAR(50) NULL,
    datatype VARCHAR(50) NULL,
    `desc` TEXT NULL,

    PRIMARY KEY (id),
    KEY idx_caracteristique_lbl (lbl)
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 4. FOURNISSEURS **** OBSOLETE VOIR PLUS HAUT
--
-- Table provisoire du module Articles.
-- Elle pourra être remplacée plus tard par le référentiel
-- organisation/entreprise.
-- =====================================================================

CREATE TABLE IF NOT EXISTS fournisseur
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
    siret VARCHAR(14) NULL,
    lbl VARCHAR(255) NOT NULL,
    adresse TEXT NULL,
    cp VARCHAR(20) NULL,
    ville VARCHAR(100) NULL,
    pays VARCHAR(100) NULL,
    contact VARCHAR(255) NULL,
    tel VARCHAR(50) NULL,
    email VARCHAR(255) NULL,
    web VARCHAR(255) NULL,
    notes TEXT NULL,

    PRIMARY KEY (id),
    KEY idx_fournisseur_lbl (lbl),
    KEY idx_fournisseur_siret (siret)
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 5. ARTICLES
--
-- Les champs métier stables restent relationnels.
--
-- data :
-- zone d'extension permettant de stocker temporairement les données
-- JSON qui seront normalisées dans de futures tables.
-- =====================================================================

CREATE TABLE IF NOT EXISTS article
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    lbl VARCHAR(255) NOT NULL,
    `desc` TEXT NULL,

    manid BIGINT UNSIGNED NULL,
    mdl VARCHAR(255) NULL,
    ean VARCHAR(32) NULL,
    typid BIGINT UNSIGNED NULL,
    ref VARCHAR(255) NULL,

    stock DECIMAL(15,3) NOT NULL DEFAULT 0,
    stockmini DECIMAL(15,3) NOT NULL DEFAULT 0,
    stockmaxi DECIMAL(15,3) NULL,

    unite VARCHAR(20) NULL,
    emplacement VARCHAR(255) NULL,
    actif TINYINT(1) NOT NULL DEFAULT 1,

    -- Extension JSON temporaire / évolutive.
    -- Exemple :
    -- {
    --     "v": 1,
    --     "caracteristiques": {},
    --     "documents": [],
    --     "metadata": {}
    -- }
    data JSON NULL,

    created_at DATETIME NULL,
    updated_at DATETIME NULL,

    PRIMARY KEY (id),

    KEY idx_article_lbl (lbl),
    KEY idx_article_ref (ref),
    KEY idx_article_ean (ean),
    KEY idx_article_manid (manid),
    KEY idx_article_typid (typid),
    KEY idx_article_actif (actif),
    KEY idx_article_stock (stock),

    CONSTRAINT fk_article_manufact
        FOREIGN KEY (manid)
        REFERENCES manufact(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE,

    CONSTRAINT fk_article_type
        FOREIGN KEY (typid)
        REFERENCES articletype(id)
        ON DELETE SET NULL
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 6. CARACTÉRISTIQUES DES ARTICLES
-- =====================================================================

CREATE TABLE IF NOT EXISTS articlecaracteristique
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    artid BIGINT UNSIGNED NOT NULL,
    carid BIGINT UNSIGNED NOT NULL,
    valeur TEXT NOT NULL,

    PRIMARY KEY (id),

    UNIQUE KEY uk_article_caracteristique (artid, carid),

    KEY idx_articlecar_artid (artid),
    KEY idx_articlecar_carid (carid),

    CONSTRAINT fk_articlecar_article
        FOREIGN KEY (artid)
        REFERENCES article(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_articlecar_caracteristique
        FOREIGN KEY (carid)
        REFERENCES caracteristique(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 7. FOURNISSEURS DES ARTICLES
-- =====================================================================

CREATE TABLE IF NOT EXISTS articlefournisseur
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    artid BIGINT UNSIGNED NOT NULL,
    fourid BIGINT UNSIGNED NOT NULL,

    ref_four VARCHAR(255) NULL,
    prix DECIMAL(15,2) NULL,
    devise VARCHAR(3) NOT NULL DEFAULT 'EUR',
    delai INT NULL,
    actif TINYINT(1) NOT NULL DEFAULT 1,

    PRIMARY KEY (id),

    KEY idx_articlefour_artid (artid),
    KEY idx_articlefour_fourid (fourid),
    KEY idx_articlefour_actif (actif),

    CONSTRAINT fk_articlefour_article
        FOREIGN KEY (artid)
        REFERENCES article(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE,

    CONSTRAINT fk_articlefour_fournisseur
        FOREIGN KEY (fourid)
        REFERENCES fournisseur(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


-- =====================================================================
-- 8. HISTORIQUE DES PRIX
-- =====================================================================

CREATE TABLE IF NOT EXISTS articleprix
(
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,

    artfid BIGINT UNSIGNED NOT NULL,
    dateprix DATE NOT NULL,
    prix DECIMAL(15,2) NOT NULL,
    devise VARCHAR(3) NOT NULL DEFAULT 'EUR',

    PRIMARY KEY (id),

    KEY idx_articleprix_artfid (artfid),
    KEY idx_articleprix_date (dateprix),

    CONSTRAINT fk_articleprix_articlefour
        FOREIGN KEY (artfid)
        REFERENCES articlefournisseur(id)
        ON DELETE CASCADE
        ON UPDATE CASCADE
)
ENGINE = InnoDB
DEFAULT CHARSET = utf8mb4
COLLATE = utf8mb4_unicode_ci;


SET FOREIGN_KEY_CHECKS = 1;
```
