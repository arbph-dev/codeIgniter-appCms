```
# cli/menus/layer5.py
"""
Couche 5 — qualification / enrichissement (stub).

Étape 1 : scan organisations type Entreprise sans extension entreprise.
WorkingMemory = liste des org à enrichir + stats.
"""
from __future__ import annotations

from rich.console import Console
from rich.prompt  import Prompt, Confirm
from rich.table   import Table
from rich.panel   import Panel

from services.auth import CredentialsStore
from services.api.organisation_client import OrganisationClient
from services.api.entreprise_client   import EntrepriseClient
from cli.menu import menu, get_auth

console = Console()

# organisation_type_id qui doivent avoir une ligne entreprises
TYPES_REQUIRING_ENTREPRISE = {1}


def menu_layer5() -> None:
    store = CredentialsStore()
    auth  = get_auth(store, "zealot")
    store.close()
    if not auth:
        return

    org_client = OrganisationClient(auth=auth)
    ent_client = EntrepriseClient(auth=auth)

    while True:
        choix = menu("Couche 5 — Enrichissement", [
            "Scan orphelines (type Entreprise sans extension)",
            "Afficher WorkingMemory",
            "Vider WorkingMemory",
            # plus tard : INSEE match, attach, push serveur
        ])
        if choix == "0":
            break
        elif choix == "1":
            _scan_orphans(org_client, ent_client)
        elif choix == "2":
            _show_wm()
        elif choix == "3":
            WorkingMemory.clear()
            console.print("[dim]WorkingMemory vidée.[/]")


# ── WorkingMemory (module-level, session CLI) ────────────────────────

class WorkingMemory:
    """Mémoire de travail couche 5 (session courante)."""
    records: list[dict] = []   # org candidates
    stats:   dict = {}

    @classmethod
    def clear(cls) -> None:
        cls.records = []
        cls.stats   = {}

    @classmethod
    def set_scan(cls, records: list[dict], scanned: int, page: int, per_page: int) -> None:
        cls.records = records
        cls.stats = {
            "scanned":  scanned,
            "to_enrich": len(records),
            "page":     page,
            "per_page": per_page,
            "ratio_pct": round(100.0 * len(records) / scanned, 1) if scanned else 0.0,
        }


def _scan_orphans(
    org_client: OrganisationClient,
    ent_client: EntrepriseClient,
) -> None:
    page     = int(Prompt.ask("Page", default="1"))
    per_page = int(Prompt.ask("Taille page", default="20"))

    # 1. Page d'organisations type Entreprise
    data = org_client.list(type_id=1, page=page, per_page=per_page)
    if not data:
        console.print("[yellow]Aucune donnée organisation.[/]")
        return

    orgs = data.get("data") or []
    pager = data.get("pager") or {}
    total_server = pager.get("total", len(orgs))

    # 2. Index des organisation_id déjà liés à une entreprise
    #    (une page d'entreprises suffit en stub ; list_all si volume faible)
    ent_page = ent_client.list(page=1, per_page=100)
    ent_items = (ent_page or {}).get("data") or []
    linked_org_ids = {
        int(e["organisation_id"])
        for e in ent_items
        if e.get("organisation_id") is not None
    }
    # Si beaucoup d'entreprises : étendre avec list_all
    if (ent_page or {}).get("pager", {}).get("total", 0) > 100:
        for e in ent_client.list_all(max_results=2000):
            if e.get("organisation_id") is not None:
                linked_org_ids.add(int(e["organisation_id"]))

    # 3. Orphelines structurelles
    orphans = []
    for org in orgs:
        oid = int(org["id"])
        type_id = int(org.get("organisation_type_id") or 0)
        if type_id not in TYPES_REQUIRING_ENTREPRISE:
            continue
        if oid not in linked_org_ids:
            orphans.append({
                "organisation_id": oid,
                "nom":             org.get("nom"),
                "siren":           org.get("siren") or None,
                "type_id":         type_id,
                "type_label":      org.get("type_label"),
                "status":          "orphan",
                "local_id":        None,   # futur SQLAlchemy
                "insee_candidates": [],
            })

    scanned = len(orgs)
    WorkingMemory.set_scan(orphans, scanned=scanned, page=page, per_page=per_page)
    st = WorkingMemory.stats

    console.print(Panel(
        f"[bold]Page {page}[/] — {scanned} orgs scannées "
        f"(total serveur type≈ {total_server})\n"
        f"[cyan]{st['to_enrich']}[/] à enrichir  "
        f"([yellow]{st['ratio_pct']} %[/] de la page)",
        title="Scan orphelines",
    ))

    if not orphans:
        console.print("[green]Aucune orpheline sur cette page.[/]")
        return

    t = Table(title="WorkingMemory — orphelines", show_lines=True)
    t.add_column("#", style="dim", width=4)
    t.add_column("org_id", style="cyan", width=8)
    t.add_column("Nom", width=40)
    t.add_column("SIREN", width=12)
    t.add_column("Type", width=12)
    for i, r in enumerate(orphans, 1):
        t.add_row(
            str(i),
            str(r["organisation_id"]),
            r["nom"] or "",
            r["siren"] or "—",
            r.get("type_label") or str(r["type_id"]),
        )
    console.print(t)
    console.print("[dim]Étape suivante : match INSEE / attach (à brancher).[/]")


def _show_wm() -> None:
    if not WorkingMemory.records:
        console.print("[dim]WorkingMemory vide.[/]")
        return
    st = WorkingMemory.stats
    console.print(
        f"scanned={st.get('scanned')}  "
        f"to_enrich={st.get('to_enrich')}  "
        f"ratio={st.get('ratio_pct')} %"
    )
    for r in WorkingMemory.records:
        console.print(
            f"  org#{r['organisation_id']}  {r['nom']!r}  "
            f"siren={r['siren'] or '—'}  status={r['status']}"
        )
```

### 5. Ordre des étapes (giveme5)

|#|Action|Statut|
|---|---|---|
|1|attach_to_organisation dans le client|ci-dessus|
|2|Scan page + WorkingMemory + %|menu layer5|
|3|Match INSEE par **nom** (q limitée) si pas de SIREN|suivant|
|4|Proposition copier / remplacer champs|suivant|
|5|attach + évent. update org (siren)|test POST org/10/entreprise|
|6|Adresse / BAN|plus tard|
|7|Note qualité pondérée par champ|plus tard|

**polymorphisme / organisation_type_id** : pour l’instant un set TYPES_REQUIRING_ENTREPRISE = {1}. Le même pattern servira pour produit_type_id → table enfant (composant, media…). Pas besoin de requêtes complexes : la règle « type X ⇒ enfant obligatoire » est déclarative dans Orbis.

---

Tu colles client + layer5.py, tu lances le scan sur une page, tu vérifies le % d’orphelines. Ensuite on branche soit le **test manuel attach** (org 10), soit le **match INSEE** sur les noms de la WorkingMemory.

**c’est exactement le rôle de la couche 5** : orchestrer 2 → 3 → 4, pas reconstruire des dicts.

### Flux correct

text

```
Zealot org (orphan)
    → EntrepriseZealot.from_api(...)          # couche 2
    → (scan) WorkingMemory[i].zealot = src

INSEE search / get
    → dict brut → EntrepriseInsee.from_api(...)   # couche 2
    → WorkingMemory[i].insee_candidates = [...]

Choix utilisateur
    → EntrepriseMapper.reconcileZealot(org, insee)  # couche 3
       ou enrichFromInsee / merge
    → EntrepriseModel                             # couche 4

Repository.upsert(model)
    → local_id dans WorkingMemory

Plus tard
    → EntrepriseClient.attach_to_organisation(id_zealot, ...)
```

Tu as déjà reconcileZealot et enrichFromInsee dans le mapper — layer5 ne fait que les appeler.

### Règle d’architecture

|Couche|Layer5 a le droit de…|
|---|---|
|2|appeler clients + Source.from_api|
|3|appeler uniquement EntrepriseMapper.*|
|4|appeler uniquement Repository / session SQLAlchemy|
|5|enchaîner, afficher, demander confirmation CLI|

Pas de construction manuelle de EntrepriseModel(...) dans le menu.

---

### Point d’attention

EntrepriseModel.siren est nullable=False.  
mapZealotToModel met siren="" si absent — l’upsert local ne doit se faire **qu’après** reconcileZealot (SIREN INSEE renseigné), sinon collision sur "".

---

En résumé : oui, réemploie EntrepriseInsee / EntrepriseZealot → EntrepriseMapper → EntrepriseModel + repository. Layer5 = chef d’orchestre + WorkingMemory + UI CLI.

Si tu as déjà EntrepriseRepository (ou équivalent dans repository.py), on peut caler le stub layer5 dessus ; sinon on esquisse un upsert_by_siren minimal.

Le repository est déjà le bon point d’entrée pour layer5.

### API utile pour la couche 5

|Méthode|Usage layer5|
|---|---|
|create(model, user=, source=)|Premier enregistrement local après reconcileZealot|
|get_by_siren(siren)|Savoir si la fiche existe déjà en local|
|update(siren, data, user=)|Enrichir une ligne existante (champs manquants)|
|get_by_id(local_id)|Relecture après save|

Pas besoin d’un upsert_by_siren dédié tout de suite : le flux suffit.

Python

```
repo = EntrepriseRepository(session)
user = {"user": "layer5", "role": "user"}

model = EntrepriseMapper.reconcileZealot(record.zealot, record.chosen_insee)

existing = repo.get_by_siren(model.siren) if model.siren else None

if existing is None:
    saved = repo.create(model, user=user, source=model.source)
else:
    # option : merge puis update champ à champ
    merged = EntrepriseMapper.merge(existing, model)  # attention: existing est ORM
    data = {
        k: getattr(merged, k)
        for k in ("denomination", "naf", "forme_juridique", "siret_siege",
                  "id_zealot", "source", "date_creation", "categorie", "etat")
        if getattr(merged, k) is not None
    }
    # id_zealot / source toujours intéressants
    data["id_zealot"] = model.id_zealot or existing.id_zealot
    saved = repo.update(model.siren, data, user=user)

record.local_id = saved.id
record.model = saved
record.status = "saved"
```

**Note** : merge() attend deux EntrepriseModel ; un objet déjà attaché à la session SQLAlchemy peut servir de base si les attributs sont les mêmes. Sinon, construire un modèle « snapshot » depuis existing.to_dict() pour éviter de muter la session trop tôt. Variante simple pour le stub :

Python

```
if existing is None:
    saved = repo.create(model, user=user, source=model.source or "insee+zealot")
else:
    # ne remplit que les trous (None côté existing)
    patch = {}
    for col in ("denomination", "sigle", "naf", "naf_naf25", "categorie", "etat",
                "forme_juridique", "nic_siege", "siret_siege", "capital",
                "tranche_effectif", "date_creation", "statut_diffusion"):
        if getattr(existing, col) is None and getattr(model, col) is not None:
            patch[col] = getattr(model, col)
    if model.id_zealot and not existing.id_zealot:
        patch["id_zealot"] = model.id_zealot
    if patch:
        saved = repo.update(model.siren, patch, user=user)
    else:
        saved = existing
```
