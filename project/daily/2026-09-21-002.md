```
import PanelBase from '/assets/js/ui/workbench/core/PanelBase.js'
import { create } from '/assets/js/core/domhelper.js'

export default class AdresseWorkbench extends PanelBase {

    constructor({ selector } = {}) {
        super()
        this._selector = selector
        this._target = null
    }

    init() {

        this._target = document.querySelector(this._selector)

        if (!this._target) {
            console.error(
                `[AdresseWorkbench] Cible introuvable : "${this._selector}"`
            )
            return this
        }

        this._render()

        return this
    }

    _render() {

        this._target.replaceChildren()

        const title = create('h3', {
            class: 'workbench-title',
            text: 'Recherche d’adresse'
        })

        const content = create('div', {
            class: 'workbench-content'

        this._target.append(title, content)

    destroy() {

        if (this._target)
            this._target.replaceChildren()

        this._target = null

        super.destroy()
    }
}
```

# 5. `WorkbenchBase` serait particulièrement utile

Je pense que c'est la prochaine bonne abstraction dans ton refactoring.

Par exemple :

```
// /assets/js/ui/workbench/core/WorkbenchBase.js

export class WorkbenchBase {

    constructor({ selector } = {}) {

        this._selector = selector
        this._target = null
    }

    init() {

        this._target = document.querySelector(this._selector)

        if (!this._target) {

            console.error(
                `[${this.constructor.name}] Cible introuvable : "${this._selector}"`
            )

            return this
        }

        this._mount()

        return this
    }

    _mount() {
        throw new Error(
            `[${this.constructor.name}] _mount() non implémenté`
        )
    }

    destroy() {

        if (this._target)
            this._target.replaceChildren()

        this._target = null
    }
}

export default WorkbenchBase
```

Ensuite :

```
export default class AdresseWorkbench extends WorkbenchBase {

    constructor() {
        super({
            selector: '#adresse-workbench'
        })
    }

    _mount() {

        this._target.append(
            ...
        )
    }
}
```

# 8. `wbapp.js` doit alors évoluer

Ton `wbapp.js` actuel contient :

```

bus.subscribe('auth:guest',   () => destroyWorkbenches())
```

Mais surtout, je ne ferais pas disparaître cette logique.

Je la déplacerais dans un gestionnaire de Workbenches.

Par exemple :

```
import { bus } from './core/eventBus.js'

import { initAuthController }
    from './features/auth/auth.controller.js'

import ToolbarAuthPanel
    from './ui/workbench/auth/ToolbarAuthPanel.js'

import AdresseWorkbench
    from './ui/workbench/adresse/AdresseWorkbench.js'


const _wbs = []

// ── Auth ─────────────────────────────────────────────────────

function initAuth() {

    initAuthController()

    new ToolbarAuthPanel().init()
}


// ── Workbenches ──────────────────────────────────────────────

function mountWorkbenches() {

    destroyWorkbenches()

    const adresse = new AdresseWorkbench({
        selector: '#adresse-workbench'
    })

    adresse.init()

    _wbs.push(adresse)
}


function destroyWorkbenches() {

    _wbs.forEach(wb => wb.destroy())

    _wbs.length = 0
}


// ── Boot ─────────────────────────────────────────────────────

function boot() {

    initAuth()

    bus.subscribe(
        'auth:success',
        () => mountWorkbenches()
    )

    bus.subscribe(
        'auth:guest',
        () => destroyWorkbenches()
    )

    bus.publish('auth:check')
}


if (document.readyState === 'loading')
    document.addEventListener('DOMContentLoaded', boot)
else
    boot()
```
