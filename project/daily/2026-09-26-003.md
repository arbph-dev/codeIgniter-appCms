# 2026-09-26-003 — Roadmap : Implémentation guideline API Tâches

## Vue d'ensemble

Implémentation d'une **guideline unifiée** pour les API backend/frontend, basée sur :
1. Patterns d'API (routes, query params, réponses standardisées)
2. Authentification renforcée (user, group, permissions)
3. Lazy loading, sparse fields, includes optionnels
4. Intégration Python/Orbis (exports/imports)

Validation progressive via deux "cas pilotes" : **GMAO/Articles** et **Tâches**.

```php
<?php

namespace App\Controllers\Api;

use App\Models\TaskModel;
use App\Traits\ApiResponse;
use CodeIgniter\RESTful\ResourceController;

class Task extends ResourceController
{
    use ApiResponse;
    protected $format = 'json';

    // ── Constantes ─────────────────────────────────────────────────────────
    
    private const ALLOWED_FIELDS = ['id', 'task_code', 'titre', 'statut', 'priorite', 'assigned_to', 'created_at'];
    private const DEFAULT_FIELDS = ['id', 'task_code', 'titre', 'statut', 'priorite', 'assigned_to'];
    private const ALLOWED_SORTS = ['task_code', 'titre', 'statut', 'priorite', 'created_at', 'date_fin_prevue'];

    // ── Helpers ────────────────────────────────────────────────────────────

    private function getModel(): TaskModel { return new TaskModel(); }

    private function resolveFields(): array
    {
        $raw = $this->request->getGet('fields') ?? '';
        if (! $raw) return self::DEFAULT_FIELDS;

        $requested = array_map('trim', explode(',', $raw));
        $valid = array_intersect($requested, self::ALLOWED_FIELDS);

        return $valid ?: self::DEFAULT_FIELDS;
    }

    private function resolveSort(): array
    {
        $sort = $this->request->getGet('sort') ?? 'task_code';
        $order = $this->request->getGet('order') ?? 'asc';

        return [
            'field' => in_array($sort, self::ALLOWED_SORTS) ? $sort : 'task_code',
            'dir' => strtolower($order) === 'desc' ? 'DESC' : 'ASC',
        ];
    }

    private function buildMeta($pager = null, array $extras = []): array
    {
        $pi = $pager ? ($pager->getDetails()['default'] ?? []) : [];

        return array_merge([
            'page' => (int) ($pi['currentPage'] ?? 1),
            'per_page' => (int) ($pi['perPage'] ?? 20),
            'total' => (int) ($pi['total'] ?? 0),
            'pages' => (int) ($pi['pageCount'] ?? 1),
        ], $extras);
    }

    // ── GET /api/task ──────────────────────────────────────────────────────

    public function index()
    {
        $q = trim($this->request->getGet('q') ?? '');
        $page = max(1, (int) ($this->request->getGet('page') ?? 1));
        $perPage = max(1, min(100, (int) ($this->request->getGet('per_page') ?? 20)));
        $fields = $this->resolveFields();
        $sort = $this->resolveSort();
        $statut = $this->request->getGet('statut');

        $model = $this->getModel()->select(implode(',', $fields));

        if ($q !== '') {
            $model->groupStart()
                ->like('task_code', $q)
                ->orLike('titre', $q)
                ->orLike('description', $q)
                ->groupEnd();
        }

        if ($statut && in_array($statut, ['todo', 'in_progress', 'done', 'blocked'])) {
            $model->where('statut', $statut);
        }

        $data = $model
            ->orderBy($sort['field'], $sort['dir'])
            ->paginate($perPage, 'default', $page);

        return $this->respond([
            'status' => 200,
            'data' => $data,
            'meta' => $this->buildMeta($model->pager, [
                'q' => $q ?: null,
                'sort' => $sort['field'],
                'order' => strtolower($sort['dir']),
                'fields' => $fields,
            ]),
        ]);
    }

    // ── GET /api/task/like ────────────────────────────────────────────────

    public function like()
    {
        $q = trim($this->request->getGet('q') ?? '');
        $len = min((int) ($this->request->getGet('len') ?? 10), 50);

        if (strlen($q) < 1) {
            return $this->respond(['status' => 200, 'data' => [], 'meta' => ['total' => 0]]);
        }

        $fields = $this->resolveFields();
        $items = $this->getModel()
            ->select(implode(',', $fields))
            ->groupStart()
                ->like('task_code', $q)
                ->orLike('titre', $q)
            ->groupEnd()
            ->orderBy('task_code', 'ASC')
            ->limit($len)
            ->find();

        $total = $this->getModel()
            ->groupStart()
                ->like('task_code', $q)
                ->orLike('titre', $q)
            ->groupEnd()
            ->countAllResults();

        return $this->respond([
            'status' => 200,
            'data' => $items,
            'meta' => ['q' => $q, 'len' => $len, 'total' => $total, 'has_more' => $total > $len],
        ]);
    }

    // ── GET /api/task/:id ──────────────────────────────────────────────────

    public function show($id = null)
    {
        $item = $this->getModel()->find((int) $id);
        if (! $item) {
            return $this->failNotFound("Tâche #{$id} introuvable.");
        }

        $include = array_filter(explode(',', $this->request->getGet('include') ?? ''));

        if (in_array('history', $include)) {
            $item['history'] = $this->db->table('task_history')
                ->where('task_id', (int) $id)
                ->orderBy('created_at', 'DESC')
                ->limit(10)
                ->get()
                ->getResultArray();
        }

        return $this->respond(['status' => 200, 'data' => $item]);
    }

    // ── POST /api/task ─────────────────────────────────────────────────────

    public function create()
    {
        $body = $this->request->getJSON(true) ?? [];
        $model = $this->getModel();

        // Génère task_code si non fourni
        if (empty($body['task_code']) && !empty($body['domaine']) && !empty($body['projet']) && !empty($body['numero'])) {
            $body['task_code'] = "{$body['domaine']}/{$body['projet']}/{$body['numero']}";
        }

        $body['created_by'] = auth()->id();

        $id = $model->insert(array_intersect_key($body, array_flip($model->allowedFields)));

        if (! $id) {
            return $this->failValidationErrors($model->errors());
        }

        return $this->respondCreated([
            'status' => 201,
            'data' => $model->find($id),
            'message' => 'Tâche créée.',
        ]);
    }

    // ── PUT /api/task/:id ──────────────────────────────────────────────────

    public function update($id = null)
    {
        $model = $this->getModel();
        $task = $model->find((int) $id);

        if (! $task) {
            return $this->failNotFound("Tâche #{$id} introuvable.");
        }

        $body = $this->request->getJSON(true) ?? [];
        $data = array_intersect_key($body, array_flip($model->allowedFields));

        // Enregistre l'historique
        $this->db->table('task_history')->insert([
            'task_id' => (int) $id,
            'modified_by' => auth()->id(),
            'action' => 'updated',
            'before_data' => json_encode($task->toArray()),
            'after_data' => json_encode($data),
            'created_at' => date('Y-m-d H:i:s'),
        ]);

        if (! $model->update((int) $id, $data)) {
            return $this->failValidationErrors($model->errors());
        }

        return $this->respond([
            'status' => 200,
            'data' => $model->find((int) $id),
            'message' => "Tâche #{$id} mise à jour.",
        ]);
    }

    // ── DELETE /api/task/:id ───────────────────────────────────────────────

    public function delete($id = null)
    {
        $model = $this->getModel();
        $task = $model->find((int) $id);

        if (! $task) {
            return $this->failNotFound("Tâche #{$id} introuvable.");
        }

        // Soft delete
        $model->delete((int) $id);

        return $this->respond([
            'status' => 200,
            'message' => "Tâche #{$id} supprimée.",
        ]);
    }
}
```

```php
<?php

namespace App\Services;

use App\Models\UserModel;
use App\Models\GroupModel;
use App\Models\PermissionModel;

class AuthService
{
    protected $userModel;
    protected $groupModel;
    protected $permModel;

    public function __construct()
    {
        $this->userModel = new UserModel();
        $this->groupModel = new GroupModel();
        $this->permModel = new PermissionModel();
    }

    /**
     * Enregistre un nouvel utilisateur
     */
    public function register(array $data): array
    {
        $validator = service('validation');
        $rules = [
            'email' => 'required|valid_email|is_unique[users.email]',
            'password' => 'required|min_length[8]',
            'name' => 'required|min_length[3]',
        ];

        if (! $validator->setRules($rules)->run($data)) {
            return ['success' => false, 'errors' => $validator->getErrors()];
        }

        $userId = $this->userModel->insert([
            'email' => $data['email'],
            'password' => password_hash($data['password'], PASSWORD_BCRYPT),
            'name' => $data['name'],
            'group_id' => $data['group_id'] ?? null,
        ]);

        return ['success' => true, 'user_id' => $userId];
    }

    /**
     * Vérifie si un utilisateur a une permission
     */
    public function hasPermission(int $userId, string $resource, string $action): bool
    {
        $user = $this->userModel->find($userId);
        if (! $user || ! $user['group_id']) {
            return false;
        }

        $permission = $this->permModel
            ->where('resource', $resource)
            ->where('action', $action)
            ->first();

        if (! $permission) {
            return false;
        }

        $groupPerm = $this->db->table('group_permissions')
            ->where('group_id', $user['group_id'])
            ->where('permission_id', $permission['id'])
            ->first();

        return (bool) $groupPerm;
    }

    /**
     * Récupère les permissions d'un groupe
     */
    public function getGroupPermissions(int $groupId): array
    {
        return $this->db->table('group_permissions')
            ->join('permissions', 'permissions.id = group_permissions.permission_id')
            ->where('group_id', $groupId)
            ->select('permissions.*')
            ->get()
            ->getResultArray();
    }
}
```

```javascript
export class TaskStore {
    constructor(apiClient) {
        this.api = apiClient;
        this.tasks = [];
        this.currentTask = null;
        this.filters = { statut: null, q: '' };
    }

    async list(page = 1, perPage = 20) {
        const params = new URLSearchParams({
            page,
            per_page: perPage,
            sort: 'task_code',
            fields: 'id,task_code,titre,statut,priorite',
            ...this.filters,
        });

        const response = await this.api.get(`/api/task?${params}`);
        this.tasks = response.data;
        return response;
    }

    async show(id, include = []) {
        const params = include.length ? `?include=${include.join(',')}` : '';
        const response = await this.api.get(`/api/task/${id}${params}`);
        this.currentTask = response.data;
        return response;
    }

    async create(task) {
        const response = await this.api.post('/api/task', task);
        this.tasks.push(response.data);
        return response;
    }

    async update(id, changes) {
        const response = await this.api.put(`/api/task/${id}`, changes);
        const idx = this.tasks.findIndex(t => t.id === id);
        if (idx >= 0) this.tasks[idx] = response.data;
        return response;
    }

    async delete(id) {
        const response = await this.api.delete(`/api/task/${id}`);
        this.tasks = this.tasks.filter(t => t.id !== id);
        return response;
    }

    async autocomplete(q) {
        const response = await this.api.get(`/api/task/like?q=${q}&len=10`);
        return response.data;
    }

    setFilter(key, value) {
        this.filters[key] = value;
    }
}
```

## Phase 5 : Intégration Orbis/Python (semaine 3)

### 5.1 Export SQL

**Fichier** : `Orbis/persistence/export_tasks.py`

```python
import json
from sqlalchemy import create_engine, text

class TaskExporter:
    def __init__(self, db_url):
        self.engine = create_engine(db_url)

    def export_to_sql(self, filename):
        """Exporte les tâches en SQL"""
        with self.engine.connect() as conn:
            result = conn.execute(text(
                "SELECT * FROM tasks WHERE deleted_at IS NULL ORDER BY created_at DESC"
            ))
            
            sqls = []
            for row in result:
                vals = ', '.join([f"'{v}'" if isinstance(v, str) else str(v) for v in row])
                sqls.append(f"INSERT INTO tasks VALUES ({vals});")
            
            with open(filename, 'w') as f:
                f.write('\n'.join(sqls))

    def export_to_json(self, filename):
        """Exporte les tâches en JSON"""
        with self.engine.connect() as conn:
            result = conn.execute(text(
                "SELECT * FROM tasks WHERE deleted_at IS NULL"
            ))
            
            tasks = [dict(row._mapping) for row in result]
            
            with open(filename, 'w') as f:
                json.dump(tasks, f, indent=2, default=str)
```
