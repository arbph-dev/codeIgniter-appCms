# 2026-09-26-004 — Roadmap : Implémentation guideline API GMAO/Articles

## Vue d'ensemble

Implémentation de l'API **GMAO/Articles** suivant la guideline établie dans 2026-09-26-003.md.

Cette API est **prioritaire** car elle sert de **cas pilote** pour valider les patterns : sparse fields, autocomplete, lazy loading, lazy relations, include optionnel, metadata enrichi.

Elle couvre les entités :
- `article` — articles de stock
- `articletype` — classification
- `caracteristique` — propriétés techniques
- `articlecaracteristique` — valeurs des propriétés (pivot)
- `articlefournisseur` — relations fournisseur + prix
- `articleprix` — historique des prix

### 2.1 Controllers

**Fichier** : `app/Controllers/Api/Article.php`

```php
<?php

namespace App\Controllers\Api;

use App\Models\ArticleModel;
use App\Traits\ApiResponse;
use CodeIgniter\RESTful\ResourceController;

class Article extends ResourceController
{
    use ApiResponse;
    protected $format = 'json';

    private const ALLOWED_FIELDS = ['id', 'lbl', 'ref', 'ean', 'typid', 'stock', 'stockmini', 'actif', 'created_at'];
    private const DEFAULT_FIELDS = ['id', 'lbl', 'ref', 'stock', 'actif'];
    private const ALLOWED_SORTS = ['lbl', 'ref', 'stock', 'created_at'];

    private function getModel(): ArticleModel { return new ArticleModel(); }

    private function resolveFields(): array
    {
        $raw = $this->request->getGet('fields') ?? '';
        if (! $raw) return self::DEFAULT_FIELDS;

        $requested = array_map('trim', explode(',', $raw));
        $valid = array_intersect($requested, self::ALLOWED_FIELDS);

        return $valid ?: self::DEFAULT_FIELDS;
    }

    private function resolveSort(): array
    {
        $sort = $this->request->getGet('sort') ?? 'lbl';
        $order = $this->request->getGet('order') ?? 'asc';

        return [
            'field' => in_array($sort, self::ALLOWED_SORTS) ? $sort : 'lbl',
            'dir' => strtolower($order) === 'desc' ? 'DESC' : 'ASC',
        ];
    }

    private function buildMeta($pager = null, array $extras = []): array
    {
        $pi = $pager ? ($pager->getDetails()['default'] ?? []) : [];

        return array_merge([
            'page' => (int) ($pi['currentPage'] ?? 1),
            'per_page' => (int) ($pi['perPage'] ?? 20),
            'total' => (int) ($pi['total'] ?? 0),
            'pages' => (int) ($pi['pageCount'] ?? 1),
        ], $extras);
    }

    // ── GET /api/article ──────────────────────────────────────────────────

    public function index()
    {
        $q = trim($this->request->getGet('q') ?? '');
        $page = max(1, (int) ($this->request->getGet('page') ?? 1));
        $perPage = max(1, min(100, (int) ($this->request->getGet('per_page') ?? 20)));
        $fields = $this->resolveFields();
        $sort = $this->resolveSort();
        $actif = $this->request->getGet('actif');

        $model = $this->getModel()->select(implode(',', $fields));

        if ($q !== '') {
            $model->groupStart()
                ->like('lbl', $q)
                ->orLike('ref', $q)
                ->orLike('ean', $q)
                ->groupEnd();
        }

        if ($actif !== null && in_array($actif, ['0', '1'])) {
            $model->where('actif', (int) $actif);
        }

        $data = $model
            ->orderBy($sort['field'], $sort['dir'])
            ->paginate($perPage, 'default', $page);

        return $this->respond([
            'status' => 200,
            'data' => $data,
            'meta' => $this->buildMeta($model->pager, [
                'q' => $q ?: null,
                'sort' => $sort['field'],
                'order' => strtolower($sort['dir']),
                'fields' => $fields,
            ]),
        ]);
    }

    // ── GET /api/article/like ────────────────────────────────────────────

    public function like()
    {
        $q = trim($this->request->getGet('q') ?? '');
        $len = min((int) ($this->request->getGet('len') ?? 10), 50);

        if (strlen($q) < 1) {
            return $this->respond(['status' => 200, 'data' => [], 'meta' => ['total' => 0]]);
        }

        $fields = $this->resolveFields();
        $items = $this->getModel()
            ->select(implode(',', $fields))
            ->where('actif', 1)
            ->groupStart()
                ->like('lbl', $q)
                ->orLike('ref', $q)
                ->orLike('ean', $q)
            ->groupEnd()
            ->orderBy('lbl', 'ASC')
            ->limit($len)
            ->find();

        $total = $this->getModel()
            ->where('actif', 1)
            ->groupStart()
                ->like('lbl', $q)
                ->orLike('ref', $q)
                ->orLike('ean', $q)
            ->groupEnd()
            ->countAllResults();

        return $this->respond([
            'status' => 200,
            'data' => $items,
            'meta' => ['q' => $q, 'len' => $len, 'total' => $total, 'has_more' => $total > $len],
        ]);
    }

    // ── GET /api/article/:id ──────────────────────────────────────────────

    public function show($id = null)
    {
        $item = $this->getModel()->find((int) $id);
        if (! $item) {
            return $this->failNotFound("Article #{$id} introuvable.");
        }

        $include = array_filter(explode(',', $this->request->getGet('include') ?? ''));

        if (in_array('caracteristiques', $include)) {
            $item['caracteristiques'] = $this->db->table('articlecaracteristique')
                ->join('caracteristique', 'caracteristique.id = articlecaracteristique.carid')
                ->select('caracteristique.*, articlecaracteristique.valeur')
                ->where('articlecaracteristique.artid', (int) $id)
                ->get()
                ->getResultArray();
        }

        if (in_array('fournisseurs', $include)) {
            $item['fournisseurs'] = $this->db->table('articlefournisseur')
                ->select('articlefournisseur.*, organisations.lbl as organisation_lbl')
                ->join('organisations', 'organisations.id = articlefournisseur.fourid')
                ->where('articlefournisseur.artid', (int) $id)
                ->where('articlefournisseur.actif', 1)
                ->get()
                ->getResultArray();
        }

        return $this->respond(['status' => 200, 'data' => $item]);
    }

    // ── POST /api/article ──────────────────────────────────────────────────

    public function create()
    {
        $body = $this->request->getJSON(true) ?? [];
        $model = $this->getModel();

        $id = $model->insert(array_intersect_key($body, array_flip($model->allowedFields)));

        if (! $id) {
            return $this->failValidationErrors($model->errors());
        }

        return $this->respondCreated([
            'status' => 201,
            'data' => $model->find($id),
            'message' => 'Article créé.',
        ]);
    }

    // ── PUT /api/article/:id ───────────────────────────────────────────────

    public function update($id = null)
    {
        $model = $this->getModel();
        if (! $model->find((int) $id)) {
            return $this->failNotFound("Article #{$id} introuvable.");
        }

        $body = $this->request->getJSON(true) ?? [];
        $data = array_intersect_key($body, array_flip($model->allowedFields));

        if (! $model->update((int) $id, $data)) {
            return $this->failValidationErrors($model->errors());
        }

        return $this->respond([
            'status' => 200,
            'data' => $model->find((int) $id),

        ]);
    }

        }
```

```python
import json
from sqlalchemy import create_engine, text
from datetime import datetime

class ArticleExporter:
    def __init__(self, db_url):
        self.engine = create_engine(db_url)

    def export_to_json(self, filename, include_suppliers=True):
        """Exporte articles + metadata en JSON"""
        with self.engine.connect() as conn:
            articles = conn.execute(text("""
                SELECT * FROM article WHERE deleted_at IS NULL
            """)).fetchall()

            data = {
                'timestamp': datetime.now().isoformat(),
                'total': len(articles),
                'articles': [dict(row._mapping) for row in articles],
            }

            if include_suppliers:
                suppliers = conn.execute(text("""
                    SELECT af.* FROM articlefournisseur af
                    JOIN article a ON a.id = af.artid
                    WHERE a.deleted_at IS NULL
                """)).fetchall()
                data['suppliers'] = [dict(row._mapping) for row in suppliers]

            with open(filename, 'w') as f:
                json.dump(data, f, indent=2, default=str)

    def export_to_sql(self, filename):
        """Exporte en SQL INSERT statements"""
        with self.engine.connect() as conn:
            result = conn.execute(text(
                "SELECT * FROM article WHERE deleted_at IS NULL"
            ))

            sqls = []
            for row in result:
                cols = ', '.join([f'`{k}`' for k in row._mapping.keys()])
                vals = ', '.join([
                    f"'{v}'" if isinstance(v, str) else str(v)
                    for v in row._mapping.values()
                ])
                sqls.append(f"INSERT INTO article ({cols}) VALUES ({vals});")

            with open(filename, 'w') as f:
                f.write('\n'.join(sqls))
```
