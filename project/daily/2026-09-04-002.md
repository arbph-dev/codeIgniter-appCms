Un script pour integrer state et service : 
```
import { ItemService } from './api/itemService.js';
import { store } from './state/store.js';

// Initialisation
document.addEventListener('DOMContentLoaded', async () => {
  const table = document.querySelector('data-table');
  const tabs = document.querySelector('custom-tabs');
  const detailsContainer = document.querySelector('#details-view');

  // 1. Écouter les événements de l'UI
  document.addEventListener('item-selected', async (e) => {
    const id = e.detail.id;
    store.selectItem(id);
    
    // Charger la suite
    const details = await ItemService.getDetails(id);
    detailsContainer.innerHTML = `<pre>${JSON.stringify(details, null, 2)}</pre>`;
    
    // Basculer automatiquement vers l'onglet détail
    tabs.selectTab('tab-details');
  });

  // 2. Écouter les changements d'état du Store
  store.addEventListener('selection-changed', (e) => {
    console.log('Nouvel élément sélectionné :', e.detail);
  });

  // 3. Charger les données initiales
  const items = await ItemService.getAll();
  table.data = items;
});
```
