3. app.js
Gère l'instanciation des composants, la gestion de la navigation, le basculement d'onglets, le mode plein écran et les thèmes.

```
document.addEventListener("DOMContentLoaded", () => {
  const pages = ["Tableau de bord", "Paramètres", "Diagnostic"];
  let currentTheme = "marine";

  const stack = document.getElementById("stack");
  const sidebarList = document.getElementById("sidebarList");
  const statusBar = document.getElementById("statusBar");
  const themeBtn = document.getElementById("themeBtn");
  const fullscreenBtn = document.getElementById("fullscreenBtn");

  // Factory pour générer le HTML d'un panneau
  function createPanel(name, index) {
    const card = document.createElement("div");
    card.className = `panel-card ${index !== 0 ? "hidden" : ""}`;
    card.dataset.index = index;

    card.innerHTML = `
      <h2 class="panel-title">${name}</h2>
      <p class="panel-description">
        Bienvenue dans le panneau ${name}. Cette zone peut contenir vos fonctions métier.
      </p>

      <div class="controls-tabs">
        <div class="tab-headers">
          <button class="tab-btn active" data-tab="controls-${index}">Contrôles disponibles</button>
          <button class="tab-btn" data-tab="info-${index}">Informations</button>
        </div>

        <div id="controls-${index}" class="tab-content active">
          <div class="form-group">
            <label>Champ texte</label>
            <input type="text" placeholder="Saisissez une valeur">
          </div>
          <div class="form-group">
            <label>Liste déroulante</label>
            <select>
              <option>Option A</option>
              <option>Option B</option>
              <option>Option C</option>
            </select>
          </div>
          <div class="form-group">
            <label>Valeur numérique</label>
            <input type="number" min="0" max="1000" step="0.01" value="0.00">
          </div>
          <div class="form-group">
            <label>
              <input type="checkbox"> Activer le contrôle
            </label>
          </div>
        </div>

        <div id="info-${index}" class="tab-content">
          <p>Cette interface peut accueillir des commandes, des indicateurs, des paramètres ou des vues de simulation.</p>
        </div>
      </div>

      <button class="primary-button switch-tab-btn">Aller au premier onglet</button>
    `;

    // Événement sur les sous-onglets
    const tabBtns = card.querySelectorAll(".tab-btn");
    tabBtns.forEach(btn => {
      btn.addEventListener("click", () => {
        card.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
        card.querySelectorAll(".tab-content").forEach(c => c.classList.remove("active"));
        
        btn.classList.add("active");
        card.querySelector(`#${btn.dataset.tab}`).classList.add("active");
      });
    });

    // Action "Aller au premier onglet"
    card.querySelector(".switch-tab-btn").addEventListener("click", () => {
      switchTab(0);
    });

    return card;
  }

  // Initialisation des panneaux dans le DOM
  pages.forEach((pageName, index) => {
    stack.appendChild(createPanel(pageName, index));
  });

  // Changement d'onglet de navigation
  function switchTab(index) {
    const items = sidebarList.querySelectorAll("li");
    const panels = stack.querySelectorAll(".panel-card");

    items.forEach((item, i) => {
      item.classList.toggle("active", i === index);
    });

    panels.forEach((panel, i) => {
      panel.classList.toggle("hidden", i !== index);
    });

    statusBar.textContent = `Onglet actif : ${pages[index]}`;
  }

  // Événement clic sur la sidebar
  sidebarList.addEventListener("click", (e) => {
    if (e.target.tagName === "LI") {
      const index = parseInt(e.target.dataset.index, 10);
      switchTab(index);
    }
  });

  // Gestion du thème
  themeBtn.addEventListener("click", () => {
    currentTheme = currentTheme === "marine" ? "nature" : "marine";
    document.documentElement.setAttribute("data-theme", currentTheme);
    themeBtn.textContent = currentTheme === "marine" ? "Thème nature" : "Thème marine";
  });

  // Gestion du plein écran
  fullscreenBtn.addEventListener("click", () => {
    if (!document.fullscreenElement) {
      document.documentElement.requestFullscreen();
    } else {
      if (document.exitFullscreen) {
        document.exitFullscreen();
      }
    }
  });
});
```

La création de Web Components (CustomElements) pour des onglets, des tableaux ou la gestion de flux de données est tout à fait pertinente. Les Web Components s'intègrent nativement dans le navigateur sans nécessiter de framework lourd.
Pour concevoir une architecture propre, maintenable et découplée, voici la méthode pas à pas.
1. Pertinence des Custom Elements
Composant Onglet (<custom-tabs>) : TRÈS PERTINENT.
Pourquoi : La logique d'activation, de masquage/affichage des panneaux et d'accessibilité (gestion du clavier) est générique et ultra-réutilisable.
Composant Tableau (<custom-table>) : PERTINENT.
Pourquoi : Idéal pour encapsuler la pagination, le tri, ou le rendu dynamique de colonnes.
Nuance : Pour du contenu totalement statique, préférez le HTML <table> natif. Conservez les Custom Elements quand il y a du dynamisme (données sous forme d'objets JavaScript).
2. Découplage de la Logique (Architecture Propre)
Pour éviter que vos composants soient fouillis ou trop couplés, séparez l'application en 4 couches distinctes :

[ Données (API / DB) ]
          ▲
          │ (Services / Fetch)
[ Gestionnaire d'État (Store) ]
          ▲
          │ (Propriétés & Événements personnalisés)
[ Composants d'UI (Onglets, Tableaux) ]


A. Accès aux données (Data Access)
Ici, aucune logique d'affichage. Utilisez des modules JS isolés (Services ou API Clients).

```

// api/itemService.js
export const ItemService = {
  async getAll() {
    const res = await fetch('/api/items');
    return res.json();
  },
  async getDetails(id) {
    const res = await fetch(`/api/items/${id}`);
    return res.json();
  }
};

```

B. Les Données et l'État (State)
​Utilisez l'API native EventTarget pour créer un Store centralisé réactif simple sans dépendances extérieures.

```
// state/store.js
class Store extends EventTarget {
  constructor() {
    super();
    this.state = {
      items: [],
      selectedItemId: null
    };
  }

  setItems(items) {
    this.state.items = items;
    this.dispatchEvent(new CustomEvent('items-changed', { detail: items }));
  }

  selectItem(id) {
    this.state.selectedItemId = id;
    this.dispatchEvent(new CustomEvent('selection-changed', { detail: id }));
  }
}

export const store = new Store();

```

C. Les Événements et l'Interaction (Event-Driven UI)
​Les Custom Elements communiquent selon une règle simple : "Props down, Events up" (Les données descendent via les propriétés JavaScript, les actions remontent via les Custom Events).
​L'élément émet un événement au clic / à la saisie.
​Le parent ou le gestionnaire écoute et met à jour l'état.
​3. Exemple d'implémentation : Système Onglet + Vue Détail
​Étape 1 : Le Composant Onglet (<custom-tabs>)
```
class CustomTabs extends HTMLElement {
  connectedCallback() {
    this.querySelectorAll('[role="tab"]').forEach(button => {
      button.addEventListener('click', () => {
        const targetId = button.getAttribute('data-target');
        this.selectTab(targetId);
      });
    });
  }

  selectTab(targetId) {
    // Masquer tous les panneaux
    this.querySelectorAll('[role="tabpanel"]').forEach(p => p.hidden = true);
    // Afficher le panneau ciblé
    const activePanel = this.querySelector(`#${targetId}`);
    if (activePanel) activePanel.hidden = false;

    // Déclencher un événement personnalisé
    this.dispatchEvent(new CustomEvent('tab-changed', { 
      detail: { tabId: targetId },
      bubbles: true 
    }));
  }
}

customElements.define('custom-tabs', CustomTabs);

```

Étape 2 : Le Tableau Interactif (<data-table>)
```
class DataTable extends HTMLElement {
  set data(items) {
    this._items = items;
    this.render();
  }

  render() {
    this.innerHTML = `
      <table>
        <tbody>
          ${this._items.map(item => `
            <tr data-id="${item.id}">
              <td>${item.name}</td>
              <td><button class="btn-select">Voir détails</button></td>
            </tr>
          `).join('')}
        </tbody>
      </table>
    `;

    this.querySelectorAll('.btn-select').forEach(btn => {
      btn.addEventListener('click', (e) => {
        const id = e.target.closest('tr').dataset.id;
        // Émettre un événement vers le haut
        this.dispatchEvent(new CustomEvent('item-selected', {
          detail: { id },
          bubbles: true
        }));
      });
    });
  }
}

customElements.define('data-table', DataTable);
```


Étape 3 : Orchestration globale (Application Principal)
​Ce contrôleur global assemble le tout sans qu'aucun composant sous-jacent ne connaisse l'existence de l'autre.

```js
import { ItemService } from './api/itemService.js';
import { store } from './state/store.js';

// Initialisation
document.addEventListener('DOMContentLoaded', async () => {
  const table = document.querySelector('data-table');
  const tabs = document.querySelector('custom-tabs');
  const detailsContainer = document.querySelector('#details-view');

  // 1. Écouter les événements de l'UI
  document.addEventListener('item-selected', async (e) => {
    const id = e.detail.id;
    store.selectItem(id);
    
    // Charger la suite
    const details = await ItemService.getDetails(id);
    detailsContainer.innerHTML = `<pre>${JSON.stringify(details, null, 2)}</pre>`;
    
    // Basculer automatiquement vers l'onglet détail
    tabs.selectTab('tab-details');
  });

  // 2. Écouter les changements d'état du Store
  store.addEventListener('selection-changed', (e) => {
    console.log('Nouvel élément sélectionné :', e.detail);
  });

  // 3. Charger les données initiales
  const items = await ItemService.getAll();
  table.data = items;
});
```


Résumé des bonnes pratiques de découplage
​Les Custom Elements restent "muets" : Ils ne font pas d'appels fetch() directement (sauf composant autonome très spécifique). Ils reçoivent des données et émettent des événements.
​Utiliser Custom Events bubbling (bubbles: true) : Permet aux composants enfants de notifier les composants parents sans dépendance directe.
​Encapsuler l'état dans un Store dédié : Évite de stocker des données applicatives complexes dans le DOM HTML lui-même.
